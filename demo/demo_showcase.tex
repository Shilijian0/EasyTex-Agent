\section{大白话转公式：矩阵乘法}
% ============================================

\textbf{用户输入：} ``写一个3x3单位矩阵右乘列向量x，等号右边是全是lambda的向量''

\textbf{Skill 生成的专业 LaTeX 代码：}

\begin{equation}
    I_3 \cdot \mathbf{x} =
    \underbrace{
        \begin{bmatrix}
            1 & 0 & 0 \\
            0 & 1 & 0 \\
            0 & 0 & 1
        \end{bmatrix}
    }_{\text{单位矩阵 } I_3}
    \cdot
    \begin{bmatrix}
        x_1 \\[2pt] x_2 \\[2pt] x_3
    \end{bmatrix}
    =
    \begin{bmatrix}
        \lambda \\[2pt] \lambda \\[2pt] \lambda
    \end{bmatrix}
    = \lambda \cdot
    \begin{bmatrix}
        1 \\[2pt] 1 \\[2pt] 1
    \end{bmatrix}
    \label{eq:identity}
\end{equation}

公式~\eqref{eq:identity} 展示了单位矩阵 $I_3$ 作用于任意向量 $\mathbf{x}$，
结果等价于将 $\mathbf{x}$ 投影到全一向量方向并缩放 $\lambda$ 倍。

\section{大白话转公式：协方差矩阵}
% ============================================

\textbf{用户输入：} ``3x3的协方差矩阵Sigma，对角线是sigma1平方、sigma2平方、sigma3平方，非对角线元素是rho-ij乘以对应的sigma''

\textbf{Skill 输出（自动加粗、编号、行列对齐）：}

\begin{equation}
    \boldsymbol{\Sigma} =
    \begin{bmatrix}
        \sigma_1^{2} & \rho_{12}\sigma_1\sigma_2 & \rho_{13}\sigma_1\sigma_3 \\[6pt]
        \rho_{21}\sigma_2\sigma_1 & \sigma_2^{2} & \rho_{23}\sigma_2\sigma_3 \\[6pt]
        \rho_{31}\sigma_3\sigma_1 & \rho_{32}\sigma_3\sigma_2 & \sigma_3^{2}
    \end{bmatrix}
    \label{eq:cov}
\end{equation}

\section{大白话转公式：贝叶斯定理}
% ============================================

\textbf{用户输入：} ``写贝叶斯定理，P(A竖线B)等于P(B竖线A)乘以P(A)除以P(B)''

\textbf{Skill 输出（自动识别条件概率符号）：}

\begin{equation}
    P(A \mid B) = \frac{P(B \mid A) \cdot P(A)}{P(B)}
    \label{eq:bayes}
\end{equation}

全概率展开形式：
\begin{equation}
    P(B) = \sum_{i=1}^{n} P(B \mid A_i) \cdot P(A_i)
    \label{eq:totalprob}
\end{equation}

\section{编译验证闭环}
% ============================================

当用户通过自然语言指令修改文档后，Skill 自动执行以下五步闭环：

\begin{enumerate}
    \item \textbf{解析导言区} — 识别已有宏包，避免重复加载和宏包冲突
    \item \textbf{字体预检查} — 查询系统字体列表，确保指定字体已安装
    \item \textbf{安全修改} — 自动备份原文件（\texttt{.bak}），原子写入新配置
    \item \textbf{静默编译} — 后台运行 \texttt{xelatex -interaction=nonstopmode}，全量捕获输出
    \item \textbf{错误过滤} — 从 \texttt{.log} 文件中精准提取错误行，过滤 overfull/underfull 等噪声
\end{enumerate}

\subsection*{致命错误拦截示例}

若用户指定了不存在的字体，Skill 的返回：

\begin{verbatim}
{
  "success": false,
  "fatal_signal": "FONT_NOT_FOUND",
  "data": {
    "message": "字体未在系统中找到，请先安装或选择替代字体",
    "font_issues": [{
      "font_name": "NotExistentFont",
      "available": false,
      "suggestions": ["Microsoft YaHei", "SimHei"]
    }]
  }
}
\end{verbatim}

\noindent \textbf{效果：} AI 收到 \texttt{FONT\_NOT\_FOUND} 信号后立即停止修改，文件未被任何改动，同时向用户提供可用的替代字体建议。
